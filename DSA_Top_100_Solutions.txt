### 1. Two Sum
PROBLEM: Return indices of two numbers that add up to target.
LOGIC: For each number, the partner we need is (target - num). Store numbers we've
seen in a map; if the partner is already there, we found the pair — one pass.
```js
function twoSum(nums, target) {
  const seen = new Map();
  for (let i = 0; i < nums.length; i++) {
    const need = target - nums[i];
    if (seen.has(need)) return [seen.get(need), i];
    seen.set(nums[i], i);
  }
}
```

### 2. Best Time to Buy & Sell Stock
PROBLEM: Max profit from one buy then one sell (buy before sell).
LOGIC: Track the lowest price so far. At each day, profit = today - minSoFar. Keep the best.
```js
function maxProfit(prices) {
  let min = Infinity, profit = 0;
  for (const p of prices) {
    min = Math.min(min, p);
    profit = Math.max(profit, p - min);
  }
  return profit;
}
```

### 3. Contains Duplicate
PROBLEM: Does the array have any duplicate value?
LOGIC: Put items in a Set; if a value is already there, it's a duplicate.
```js
function containsDuplicate(nums) {
  const set = new Set();
  for (const n of nums) { if (set.has(n)) return true; set.add(n); }
  return false;
}
```

### 4. Valid Anagram
PROBLEM: Is string t a rearrangement of s?
LOGIC: Same letters with same counts. Count letters of s, subtract using t; all zero = anagram.
```js
function isAnagram(s, t) {
  if (s.length !== t.length) return false;
  const c = {};
  for (const ch of s) c[ch] = (c[ch] || 0) + 1;
  for (const ch of t) { if (!c[ch]) return false; c[ch]--; }
  return true;
}
```

### 5. Group Anagrams
PROBLEM: Group words that are anagrams of each other.
LOGIC: Anagrams share the same sorted string (key). Bucket words by that key in a map.
```js
function groupAnagrams(strs) {
  const map = {};
  for (const w of strs) {
    const key = [...w].sort().join("");
    (map[key] ||= []).push(w);
  }
  return Object.values(map);
}
```

### 6. Top K Frequent Elements
PROBLEM: Return the k most frequent numbers.
LOGIC: Count frequencies, then bucket by frequency (index = count). Read buckets from high to low.
```js
function topKFrequent(nums, k) {
  const count = {};
  for (const n of nums) count[n] = (count[n] || 0) + 1;
  const buckets = Array(nums.length + 1).fill().map(() => []);
  for (const n in count) buckets[count[n]].push(+n);
  const res = [];
  for (let i = buckets.length - 1; i >= 0 && res.length < k; i--) res.push(...buckets[i]);
  return res.slice(0, k);
}
```

### 7. Product of Array Except Self
PROBLEM: output[i] = product of all elements except nums[i] (no division).
LOGIC: Multiply everything to the left, then everything to the right, in two passes.
```js
function productExceptSelf(nums) {
  const res = Array(nums.length).fill(1);
  let pre = 1;
  for (let i = 0; i < nums.length; i++) { res[i] = pre; pre *= nums[i]; }
  let post = 1;
  for (let i = nums.length - 1; i >= 0; i--) { res[i] *= post; post *= nums[i]; }
  return res;
}
```

### 8. Encode & Decode Strings
PROBLEM: Serialize a list of strings into one string and back.
LOGIC: Prefix each string with its length + a delimiter, so decoding knows exactly how many chars to read.
```js
function encode(strs) { return strs.map(s => s.length + "#" + s).join(""); }
function decode(str) {
  const res = []; let i = 0;
  while (i < str.length) {
    let j = i; while (str[j] !== "#") j++;
    const len = +str.slice(i, j);
    res.push(str.slice(j + 1, j + 1 + len));
    i = j + 1 + len;
  }
  return res;
}
```

### 9. Longest Consecutive Sequence
PROBLEM: Length of longest run of consecutive integers (unsorted).
LOGIC: Put all in a Set. Only start counting at a number with no left neighbor (n-1 missing), then count up.
```js
function longestConsecutive(nums) {
  const set = new Set(nums); let best = 0;
  for (const n of set) {
    if (!set.has(n - 1)) {              // sequence start
      let len = 1;
      while (set.has(n + len)) len++;
      best = Math.max(best, len);
    }
  }
  return best;
}
```

### 10. Maximum Subarray (Kadane's)
PROBLEM: Largest sum of a contiguous subarray.
LOGIC: Walk through; either extend the current subarray or start fresh at the current number. Track best.
```js
function maxSubArray(nums) {
  let cur = nums[0], best = nums[0];
  for (let i = 1; i < nums.length; i++) {
    cur = Math.max(nums[i], cur + nums[i]);
    best = Math.max(best, cur);
  }
  return best;
}
```

### 11. Maximum Product Subarray
PROBLEM: Largest product of a contiguous subarray.
LOGIC: A negative can flip min↔max, so track both running max and min; swap on negatives.
```js
function maxProduct(nums) {
  let max = nums[0], min = nums[0], res = nums[0];
  for (let i = 1; i < nums.length; i++) {
    if (nums[i] < 0) [max, min] = [min, max];
    max = Math.max(nums[i], max * nums[i]);
    min = Math.min(nums[i], min * nums[i]);
    res = Math.max(res, max);
  }
  return res;
}
```

### 12. Valid Palindrome
PROBLEM: Is the string a palindrome (ignore case/non-alphanumeric)?
LOGIC: Clean, then compare from both ends inward.
```js
function isPalindrome(s) {
  s = s.toLowerCase().replace(/[^a-z0-9]/g, "");
  let l = 0, r = s.length - 1;
  while (l < r) if (s[l++] !== s[r--]) return false;
  return true;
}
```

### 13. Two Sum II (sorted)
PROBLEM: Two numbers in a sorted array summing to target (1-indexed).
LOGIC: Move left/right pointers; if sum too big shrink right, too small grow left.
```js
function twoSumII(nums, target) {
  let l = 0, r = nums.length - 1;
  while (l < r) {
    const s = nums[l] + nums[r];
    if (s === target) return [l + 1, r + 1];
    s < target ? l++ : r--;
  }
}
```

### 14. 3Sum
PROBLEM: All unique triplets that sum to 0.
LOGIC: Sort. Fix one number, two-pointer the rest for the remaining target; skip duplicates.
```js
function threeSum(nums) {
  nums.sort((a, b) => a - b);
  const res = [];
  for (let i = 0; i < nums.length - 2; i++) {
    if (i > 0 && nums[i] === nums[i - 1]) continue;
    let l = i + 1, r = nums.length - 1;
    while (l < r) {
      const s = nums[i] + nums[l] + nums[r];
      if (s < 0) l++;
      else if (s > 0) r--;
      else {
        res.push([nums[i], nums[l], nums[r]]);
        while (l < r && nums[l] === nums[l + 1]) l++;
        while (l < r && nums[r] === nums[r - 1]) r--;
        l++; r--;
      }
    }
  }
  return res;
}
```

### 15. Container With Most Water
PROBLEM: Max water between two vertical lines.
LOGIC: Start widest. Area = width × min(height). Move the shorter wall inward (only that can help).
```js
function maxArea(height) {
  let l = 0, r = height.length - 1, best = 0;
  while (l < r) {
    best = Math.max(best, (r - l) * Math.min(height[l], height[r]));
    height[l] < height[r] ? l++ : r--;
  }
  return best;
}
```

### 16. Trapping Rain Water
PROBLEM: Water trapped between bars.
LOGIC: Water above a bar = min(maxLeft, maxRight) - height. Two pointers track running maxes.
```js
function trap(height) {
  let l = 0, r = height.length - 1, lMax = 0, rMax = 0, res = 0;
  while (l < r) {
    if (height[l] < height[r]) {
      lMax = Math.max(lMax, height[l]); res += lMax - height[l]; l++;
    } else {
      rMax = Math.max(rMax, height[r]); res += rMax - height[r]; r--;
    }
  }
  return res;
}
```

### 17. Longest Substring Without Repeating Characters
PROBLEM: Length of longest substring with all unique chars.
LOGIC: Grow a window; when a repeat appears, jump the window start past the previous occurrence.
```js
function lengthOfLongestSubstring(s) {
  const last = new Map(); let start = 0, best = 0;
  for (let i = 0; i < s.length; i++) {
    if (last.has(s[i]) && last.get(s[i]) >= start) start = last.get(s[i]) + 1;
    last.set(s[i], i);
    best = Math.max(best, i - start + 1);
  }
  return best;
}
```

### 18. Longest Repeating Character Replacement
PROBLEM: Longest substring of one repeating letter if you can replace up to k chars.
LOGIC: Window is valid if (windowLen - countOfMostFrequentChar) ≤ k. Shrink when invalid.
```js
function characterReplacement(s, k) {
  const count = {}; let l = 0, maxF = 0, best = 0;
  for (let r = 0; r < s.length; r++) {
    count[s[r]] = (count[s[r]] || 0) + 1;
    maxF = Math.max(maxF, count[s[r]]);
    while (r - l + 1 - maxF > k) count[s[l++]]--;
    best = Math.max(best, r - l + 1);
  }
  return best;
}
```

### 19. Permutation in String
PROBLEM: Does s2 contain a permutation of s1?
LOGIC: Slide a fixed window of size |s1| over s2; compare character counts.
```js
function checkInclusion(s1, s2) {
  if (s1.length > s2.length) return false;
  const need = Array(26).fill(0), win = Array(26).fill(0);
  const idx = (c) => c.charCodeAt(0) - 97;
  for (let i = 0; i < s1.length; i++) { need[idx(s1[i])]++; win[idx(s2[i])]++; }
  const eq = () => need.every((v, i) => v === win[i]);
  if (eq()) return true;
  for (let i = s1.length; i < s2.length; i++) {
    win[idx(s2[i])]++; win[idx(s2[i - s1.length])]--;
    if (eq()) return true;
  }
  return false;
}
```

### 20. Minimum Window Substring
PROBLEM: Smallest substring of s containing all chars of t.
LOGIC: Expand right to cover all needed chars, then shrink left while still valid; track smallest.
```js
function minWindow(s, t) {
  if (!t || !s) return "";
  const need = {}; for (const c of t) need[c] = (need[c] || 0) + 1;
  let required = Object.keys(need).length, formed = 0;
  const win = {}; let l = 0, best = [Infinity, 0, 0];
  for (let r = 0; r < s.length; r++) {
    const c = s[r]; win[c] = (win[c] || 0) + 1;
    if (need[c] && win[c] === need[c]) formed++;
    while (formed === required) {
      if (r - l + 1 < best[0]) best = [r - l + 1, l, r];
      const lc = s[l++]; win[lc]--;
      if (need[lc] && win[lc] < need[lc]) formed--;
    }
  }
  return best[0] === Infinity ? "" : s.slice(best[1], best[2] + 1);
}
```

### 21. Sliding Window Maximum
PROBLEM: Max of every window of size k.
LOGIC: Keep a deque of indices in decreasing value order; front is always the window max.
```js
function maxSlidingWindow(nums, k) {
  const dq = [], res = [];
  for (let i = 0; i < nums.length; i++) {
    while (dq.length && nums[dq[dq.length - 1]] < nums[i]) dq.pop();
    dq.push(i);
    if (dq[0] <= i - k) dq.shift();
    if (i >= k - 1) res.push(nums[dq[0]]);
  }
  return res;
}
```

### 22. Valid Parentheses
PROBLEM: Are brackets balanced/correctly nested?
LOGIC: Push opens; on a close, the top must be its matching open.
```js
function isValid(s) {
  const map = { ")": "(", "]": "[", "}": "{" }, st = [];
  for (const c of s) {
    if (c in map) { if (st.pop() !== map[c]) return false; }
    else st.push(c);
  }
  return st.length === 0;
}
```

### 23. Min Stack
PROBLEM: Stack with O(1) getMin.
LOGIC: Keep a second stack tracking the minimum at each level.
```js
class MinStack {
  constructor() { this.s = []; this.m = []; }
  push(x) { this.s.push(x); this.m.push(this.m.length ? Math.min(x, this.top(this.m)) : x); }
  pop() { this.s.pop(); this.m.pop(); }
  top(a = this.s) { return a[a.length - 1]; }
  getMin() { return this.m[this.m.length - 1]; }
}
```

### 24. Evaluate Reverse Polish Notation
PROBLEM: Evaluate postfix expression.
LOGIC: Push numbers; on an operator pop two, apply, push result.
```js
function evalRPN(tokens) {
  const st = [], ops = { "+": (a, b) => a + b, "-": (a, b) => a - b, "*": (a, b) => a * b, "/": (a, b) => (a / b) | 0 };
  for (const t of tokens) {
    if (t in ops) { const b = st.pop(), a = st.pop(); st.push(ops[t](a, b)); }
    else st.push(+t);
  }
  return st[0];
}
```

### 25. Generate Parentheses
PROBLEM: All valid combinations of n pairs of parentheses.
LOGIC: Backtrack; add '(' if opens left, ')' if it won't break validity (closes < opens).
```js
function generateParenthesis(n) {
  const res = [];
  function bt(s, open, close) {
    if (s.length === 2 * n) { res.push(s); return; }
    if (open < n) bt(s + "(", open + 1, close);
    if (close < open) bt(s + ")", open, close + 1);
  }
  bt("", 0, 0);
  return res;
}
```

### 26. Daily Temperatures
PROBLEM: Days to wait for a warmer temperature.
LOGIC: Monotonic stack of indices; when a warmer day comes, resolve all colder days on the stack.
```js
function dailyTemperatures(T) {
  const res = Array(T.length).fill(0), st = [];
  for (let i = 0; i < T.length; i++) {
    while (st.length && T[i] > T[st[st.length - 1]]) {
```

### 27. Largest Rectangle in Histogram
PROBLEM: Largest rectangle area in a bar chart.
LOGIC: Increasing stack of indices; when a shorter bar appears, pop and compute areas with each popped bar as the smallest.
```js
function largestRectangleArea(heights) {
  const st = []; let best = 0;
  heights.push(0); // sentinel
  for (let i = 0; i < heights.length; i++) {
    while (st.length && heights[i] < heights[st[st.length - 1]]) {
      const h = heights[st.pop()];
      const w = st.length ? i - st[st.length - 1] - 1 : i;
```

### 28. Binary Search
PROBLEM: Find target index in a sorted array.
LOGIC: Halve the search range each step by comparing to the middle.
```js
function search(nums, target) {
  let lo = 0, hi = nums.length - 1;
  while (lo <= hi) {
    const mid = (lo + hi) >> 1;
    if (nums[mid] === target) return mid;
    nums[mid] < target ? lo = mid + 1 : hi = mid - 1;
  }
  return -1;
}
```

### 29. Search a 2D Matrix
PROBLEM: Search a value in a row-and-column sorted matrix.
LOGIC: Treat the matrix as one sorted array of size m*n and binary search; map index→(row,col).
```js
function searchMatrix(matrix, target) {
  const m = matrix.length, n = matrix[0].length;
  let lo = 0, hi = m * n - 1;
  while (lo <= hi) {
    const mid = (lo + hi) >> 1, val = matrix[(mid / n) | 0][mid % n];
    if (val === target) return true;
    val < target ? lo = mid + 1 : hi = mid - 1;
  }
  return false;
}
```

### 30. Koko Eating Bananas
PROBLEM: Min eating speed to finish piles within h hours.
LOGIC: Answer is a speed between 1 and max pile. Binary search the speed; check feasibility.
```js
function minEatingSpeed(piles, h) {
  const hours = (k) => piles.reduce((s, p) => s + Math.ceil(p / k), 0);
  let lo = 1, hi = Math.max(...piles);
  while (lo < hi) {
    const mid = (lo + hi) >> 1;

}
```

### 31. Find Minimum in Rotated Sorted Array
PROBLEM: Min of a rotated sorted array.
LOGIC: If mid > right, min is to the right; else it's mid or left.
```js
function findMin(nums) {

  while (lo < hi) {
    const mid = (lo + hi) >> 1;
    nums[mid] > nums[hi] ? lo = mid + 1 : hi = mid;
  }
  return nums[lo];
}
```

### 32. Search in Rotated Sorted Array
PROBLEM: Find target in a rotated sorted array.
LOGIC: One half is always sorted. Check which half is sorted and whether target lies in it.
```js
function searchRotated(nums, target) {
  let lo = 0, hi = nums.length - 1;
  while (lo <= hi) {
    const mid = (lo + hi) >> 1;
    if (nums[mid] === target) return mid;
    if (nums[lo] <= nums[mid]) {              // left half sorted
      if (nums[lo] <= target && target < nums[mid]) hi = mid - 1; else lo = mid + 1;
    } else {                                  // right half sorted
      if (nums[mid] < target && target <= nums[hi]) lo = mid + 1; else hi = mid - 1;
    }
  }
  return -1;
}
```

### 33. Median of Two Sorted Arrays
PROBLEM: Median of two sorted arrays in O(log).
LOGIC: Binary search a partition of the smaller array so left halves ≤ right halves.
```js
function findMedianSortedArrays(a, b) {
  if (a.length > b.length) [a, b] = [b, a];
  const m = a.length, n = b.length, half = (m + n + 1) >> 1;
  let lo = 0, hi = m;
  while (lo <= hi) {
    const i = (lo + hi) >> 1, j = half - i;
    const aL = i ? a[i - 1] : -Infinity, aR = i < m ? a[i] : Infinity;
    const bL = j ? b[j - 1] : -Infinity, bR = j < n ? b[j] : Infinity;
    if (aL <= bR && bL <= aR) {
      if ((m + n) % 2) return Math.max(aL, bL);
      return (Math.max(aL, bL) + Math.min(aR, bR)) / 2;
    }
    aL > bR ? hi = i - 1 : lo = i + 1;
  }
}
```

### 34. Reverse Linked List
LOGIC: Flip each node's next pointer to the previous node.
```js
function reverseList(head) {
  let prev = null;
  while (head) { const next = head.next; head.next = prev; prev = head; head = next; }
  return prev;
}
```

### 35. Merge Two Sorted Lists
LOGIC: Use a dummy head; repeatedly attach the smaller current node.
```js
function mergeTwoLists(a, b) {
  const dummy = { next: null }; let tail = dummy;
  while (a && b) { if (a.val <= b.val) { tail.next = a; a = a.next; } else { tail.next = b; b = b.next; } tail = tail.next; }
  tail.next = a || b;
  return dummy.next;
}
```

### 36. Linked List Cycle
LOGIC: Fast pointer moves 2, slow moves 1; they meet if there's a loop.
```js
function hasCycle(head) {
  let slow = head, fast = head;
  while (fast && fast.next) { slow = slow.next; fast = fast.next.next; if (slow === fast) return true; }
  return false;
}
```

### 37. Reorder List
LOGIC: Find middle, reverse the second half, then merge the two halves alternately.
```js
function reorderList(head) {
  let slow = head, fast = head;
  while (fast.next && fast.next.next) { slow = slow.next; fast = fast.next.next; }
  let second = slow.next; slow.next = null; let prev = null;
  while (second) { const n = second.next; second.next = prev; prev = second; second = n; }
  let first = head; second = prev;
  while (second) { const n1 = first.next, n2 = second.next; first.next = second; second.next = n1; first = n1; second = n2; }
}
```

### 38. Remove Nth Node From End
LOGIC: Advance a lead pointer n steps, then move both until lead hits the end.
```js
function removeNthFromEnd(head, n) {
  const dummy = { next: head }; let lead = dummy, trail = dummy;
  for (let i = 0; i < n; i++) lead = lead.next;
  while (lead.next) { lead = lead.next; trail = trail.next; }
  trail.next = trail.next.next;
  return dummy.next;
}
```

### 39. Copy List with Random Pointer
LOGIC: Map each original node to its clone, then wire next/random using the map.
```js
function copyRandomList(head) {
  const map = new Map();
  let cur = head;
  while (cur) { map.set(cur, { val: cur.val }); cur = cur.next; }
  cur = head;
  while (cur) { const c = map.get(cur); c.next = map.get(cur.next) || null; c.random = map.get(cur.random) || null; cur = cur.next; }
  return map.get(head) || null;
}
```

### 40. Add Two Numbers
LOGIC: Add digit-by-digit (lists are reversed), carrying over to the next node.
```js
function addTwoNumbers(a, b) {
  const dummy = { next: null }; let tail = dummy, carry = 0;
  while (a || b || carry) {
    const sum = (a?.val || 0) + (b?.val || 0) + carry;
    carry = (sum / 10) | 0;
    tail.next = { val: sum % 10, next: null }; tail = tail.next;
    a = a?.next; b = b?.next;
  }
  return dummy.next;
}
```

### 41. LRU Cache
LOGIC: JS Map keeps insertion order. On get/put, delete+reinsert to mark "recent"; evict the first (oldest) key when full.
```js
class LRUCache {
  constructor(cap) { this.cap = cap; this.map = new Map(); }
  get(key) {
    if (!this.map.has(key)) return -1;
    const v = this.map.get(key); this.map.delete(key); this.map.set(key, v);
    return v;
  }
  put(key, val) {
    if (this.map.has(key)) this.map.delete(key);
    this.map.set(key, val);
    if (this.map.size > this.cap) this.map.delete(this.map.keys().next().value);
  }
}
```

### 42. Merge K Sorted Lists
LOGIC: Repeatedly merge lists in pairs (divide & conquer) — O(n log k).
```js
function mergeKLists(lists) {
  if (!lists.length) return null;
  while (lists.length > 1) {
    const merged = [];
    for (let i = 0; i < lists.length; i += 2)
      merged.push(mergeTwoLists(lists[i], lists[i + 1] || null));
    lists = merged;
  }
  return lists[0];
}
```

### 43. Invert Binary Tree
LOGIC: Swap left/right at every node recursively.
```js
function invertTree(root) {
  if (!root) return null;
  [root.left, root.right] = [invertTree(root.right), invertTree(root.left)];

}
```

### 44. Maximum Depth of Binary Tree
LOGIC: Depth = 1 + max(depth of left, depth of right).

### 46. Balanced Binary Tree

### 67. Combination Sum
LOGIC: Try each candidate (reuse allowed); subtract from target; backtrack when target<0 or solved.

```js
function combinationSum(candidates, target) {
```

### 72. N-Queens

```js
function solveNQueens(n) {
```

### 84. Network Delay Time (Dijkstra)

```js
function networkDelayTime(times, n, k) {

  const dist = {}; const pq = [[0, k]];
```

### 85. Merge Intervals
LOGIC: Sort by start; if the next interval overlaps the last merged one, extend it.
```js
function merge(intervals) {
  intervals.sort((a, b) => a[0] - b[0]);
  const res = [intervals[0]];
  for (const [s, e] of intervals.slice(1)) {
    const last = res[res.length - 1];
    if (s <= last[1]) last[1] = Math.max(last[1], e);
    else res.push([s, e]);
  }
  return res;
}
```

### 86. Insert Interval
LOGIC: Add intervals ending before the new one, merge overlaps, then add the rest.
```js
function insert(intervals, newInterval) {
  const res = []; let i = 0, n = intervals.length;
  while (i < n && intervals[i][1] < newInterval[0]) res.push(intervals[i++]);
  while (i < n && intervals[i][0] <= newInterval[1]) {
    newInterval[0] = Math.min(newInterval[0], intervals[i][0]);
    newInterval[1] = Math.max(newInterval[1], intervals[i][1]); i++;
  }
  res.push(newInterval);
  while (i < n) res.push(intervals[i++]);
  return res;
}
```

### 87. Non-overlapping Intervals
LOGIC: Sort by end time; greedily keep intervals that don't overlap; count removals.
```js
function eraseOverlapIntervals(intervals) {
  intervals.sort((a, b) => a[1] - b[1]);
  let end = -Infinity, removed = 0;
  for (const [s, e] of intervals) {
    if (s >= end) end = e; else removed++;
  }
  return removed;
}
```

### 88. Meeting Rooms II
LOGIC: Count max overlapping meetings = rooms. Sort starts and ends; sweep with two pointers.
```js
function minMeetingRooms(intervals) {
  const starts = intervals.map(i => i[0]).sort((a, b) => a - b);
  const ends = intervals.map(i => i[1]).sort((a, b) => a - b);
  let rooms = 0, max = 0, s = 0, e = 0;
  while (s < starts.length) {
    if (starts[s] < ends[e]) { rooms++; s++; max = Math.max(max, rooms); }
    else { rooms--; e++; }
  }
  return max;
}
```

### 89. Jump Game
LOGIC: Track the farthest index reachable; if you ever can't move forward, you fail.
```js
function canJump(nums) {
  let reach = 0;
  for (let i = 0; i < nums.length; i++) {
    if (i > reach) return false;
    reach = Math.max(reach, i + nums[i]);
  }
  return true;
}
```

### 90. Climbing Stairs
LOGIC: Ways to reach step i = ways(i-1) + ways(i-2) (Fibonacci).
```js
function climbStairs(n) {
  let a = 1, b = 1;
  for (let i = 0; i < n; i++) [a, b] = [b, a + b];
  return a;
}
```

### 91. House Robber
LOGIC: At each house, max(rob it + best two back, skip it = best one back).
```js
function rob(nums) {
  let prev = 0, cur = 0;
  for (const n of nums) [prev, cur] = [cur, Math.max(cur, prev + n)];
  return cur;
}
```

### 92. House Robber II (circular)
LOGIC: Houses are in a circle, so either rob houses [0..n-2] or [1..n-1]; take the better.
```js
function robII(nums) {
  if (nums.length === 1) return nums[0];
  const lin = (arr) => { let p = 0, c = 0; for (const n of arr) [p, c] = [c, Math.max(c, p + n)]; return c; };
  return Math.max(lin(nums.slice(1)), lin(nums.slice(0, -1)));
}
```

### 93. Coin Change
LOGIC: dp[a] = fewest coins to make amount a. Try each coin: dp[a] = min(dp[a], dp[a-coin]+1).
```js
function coinChange(coins, amount) {
  const dp = Array(amount + 1).fill(Infinity); dp[0] = 0;
  for (const c of coins) for (let a = c; a <= amount; a++) dp[a] = Math.min(dp[a], dp[a - c] + 1);
  return dp[amount] === Infinity ? -1 : dp[amount];
}
```

### 94. Longest Increasing Subsequence
LOGIC: Keep a "tails" array; binary-search where each number fits — its length is the LIS.
```js
function lengthOfLIS(nums) {
  const tails = [];
  for (const n of nums) {
    let lo = 0, hi = tails.length;
    while (lo < hi) { const m = (lo + hi) >> 1; tails[m] < n ? lo = m + 1 : hi = m; }
    tails[lo] = n;
  }
  return tails.length;
}
```

### 95. Word Break
LOGIC: dp[i] = can the prefix of length i be segmented? True if some j<i where dp[j] and s[j..i] is a word.
```js
function wordBreak(s, wordDict) {
  const set = new Set(wordDict), dp = Array(s.length + 1).fill(false); dp[0] = true;
  for (let i = 1; i <= s.length; i++)
    for (let j = 0; j < i; j++)
      if (dp[j] && set.has(s.slice(j, i))) { dp[i] = true; break; }
  return dp[s.length];
}
```

### 96. Decode Ways
LOGIC: dp[i] = ways to decode first i chars. Add dp[i-1] if last digit valid (1-9), dp[i-2] if last two form 10-26.
```js
function numDecodings(s) {
  const n = s.length, dp = Array(n + 1).fill(0);
  dp[0] = 1; dp[1] = s[0] === "0" ? 0 : 1;
  for (let i = 2; i <= n; i++) {
    if (s[i - 1] !== "0") dp[i] += dp[i - 1];
    const two = +s.slice(i - 2, i);
    if (two >= 10 && two <= 26) dp[i] += dp[i - 2];
  }
  return dp[n];
}
```

### 97. Unique Paths
LOGIC: Ways to reach a cell = ways from above + ways from left. Fill a grid.
```js
function uniquePaths(m, n) {
  const dp = Array(n).fill(1);
  for (let i = 1; i < m; i++) for (let j = 1; j < n; j++) dp[j] += dp[j - 1];
  return dp[n - 1];
}
```

### 98. Longest Common Subsequence
LOGIC: If chars match, dp[i][j]=dp[i-1][j-1]+1; else max(drop one char from either string).
```js
function longestCommonSubsequence(a, b) {
  const dp = Array.from({ length: a.length + 1 }, () => Array(b.length + 1).fill(0));
  for (let i = 1; i <= a.length; i++)
    for (let j = 1; j <= b.length; j++)
      dp[i][j] = a[i-1] === b[j-1] ? dp[i-1][j-1] + 1 : Math.max(dp[i-1][j], dp[i][j-1]);
  return dp[a.length][b.length];
}
```

### 99. Edit Distance
LOGIC: Min insert/delete/replace to turn a→b. If chars match no cost; else 1 + min(3 operations).
```js
function minDistance(a, b) {
  const dp = Array.from({ length: a.length + 1 }, (_, i) => Array(b.length + 1).fill(0).map((_, j) => i === 0 ? j : j === 0 ? i : 0));
  for (let i = 1; i <= a.length; i++)
    for (let j = 1; j <= b.length; j++)
      dp[i][j] = a[i-1] === b[j-1] ? dp[i-1][j-1] : 1 + Math.min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1]);
  return dp[a.length][b.length];
}
```

### 100. Partition Equal Subset Sum (0/1 Knapsack)
LOGIC: Can we pick a subset summing to total/2? Subset-sum DP over reachable sums.
```js
function canPartition(nums) {
  const sum = nums.reduce((a, b) => a + b, 0);
  if (sum % 2) return false;
  const target = sum / 2, dp = new Set([0]);
  for (const n of nums) {
    const next = new Set(dp);
    for (const s of dp) if (s + n <= target) next.add(s + n);
    if (next.has(target)) return true;
    dp.clear(); next.forEach(v => dp.add(v));
  }
  return dp.has(target);
}
```
